\newcommand{\sphToCart}[3]
        {
          \def\rpar{#1}
          \def\thetapar{#2}
          \def\phipar{#3}

          \pgfmathsetmacro{\x}{\rpar*sin(\thetapar)*cos(\phipar)}
          \pgfmathsetmacro{\y}{\rpar*sin(\thetapar)*sin(\phipar)}
          \pgfmathsetmacro{\z}{\rpar*cos(\thetapar)}
        }
